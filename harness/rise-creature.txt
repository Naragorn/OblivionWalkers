ini Hands Enabled=1
ini Debug GripTrace=0
console-at ready player.placeatme 00069AD1 1
console-at ready tgm
console-at placed player.setangle z 0
console-at placed {actor}.moveto player 0 150 0
console-at moved killall

expect HandScript: finished
expect 0 loaded, 0 dead: none
expect 1 loaded, 1 dead: 00069AD1 x1 (dead 1)
expect 00080
count 2 00080
count 1 x1 (dead 1)
reject .resurrect
reject HandScript: WARNING

wait 12
right pos 0.20 -0.40 -0.30
left pos -0.20 -0.40 -0.30
wait 1
mark ready
wait 6
mark placed
wait 1
mark moved
wait 3
mark dead
wait 32
mark risen
wait 4
mark risen2
