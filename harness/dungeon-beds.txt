# The beds of a refuge (docs/walkers-spec.md 4.20.7). tools/dungeon-walkers.sh
# beds sets [Dungeons] TestSite=Fort Ash, TestSiteState=1 (a refuge), five
# people, Beds=3.
# - outside, then inside: "beds - bed 0 laid in Fort Ash on spot 11", bed 1
#   on 10, bed 2 on 9 - one a second;
# - bed0, bed1: the player put 110 units from a bed, looking at it (tlb: the
#   place lit) - the eye pictures;
# - sleep: bed 0 activated for the player - the game's sleep menu ("menu
#   SleepWait" in OBVR's log); closed again;
# - off: the dungeons turned off (LINK's variable) - "beds - 3 taken away".
# Run through tools/dungeon-walkers.sh beds.

ini Hands Enabled=1
ini Debug GripTrace=0
ini Debug EyeDump=1
ini Performance Enabled=0
ini SelfCheck ShowPanel=0
console-at ready tgm
console-at ready tdetect
console-at ready set timescale to 0
console-at ready cow Tamriel -9 19
console-at inside coc FortAsh01
console-at seen CloseAllMenus
console-at seen tlb
console-at look0 player.moveto {form:OblivionWalkers.esp:A8F} 110 0 20
console-at look0 player.setangle z 270
console-at look0 player.setangle x 25
console-at bed0 player.moveto {form:OblivionWalkers.esp:A90} 0 110 20
console-at bed0 player.setangle z 180
console-at bed0 player.setangle x 25
console-at bed1 "{form:OblivionWalkers.esp:A8F}".activate player
console-at sleep CloseAllMenus
console-at sleep set Walkers.ini_dungeons_enabled to 0

expect HandScript: finished
reject HandScript: WARNING

wait 12
right pos 0.20 -0.40 -0.30
left pos -0.20 -0.40 -0.30
wait 1
mark ready
wait 16
mark outside
wait 2
mark inside
wait 6
mark seen
wait 6
mark look0
wait 3
mark bed0
wait 3
mark bed1
wait 4
mark sleep
wait 6
mark end
wait 1
